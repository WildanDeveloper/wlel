enum Shape {
    Circle(float),
    Rect(float, float),
    Point,
}

fn area(s: Shape) -> float {
    return match s {
        Circle(r) => 3.141592653589793 * r * r,
        Rect(w, h) => w * h,
        Point => 0.0,
    };
}

fn log_shape(s: Shape) {
    match s {
        Circle(r) => {
            std::println_float(r);
        }
        Rect(w, h) => {
            std::println_float(w + h);
        }
        Point => {
        }
    }
}
