extern fn c_abs(x: int) -> int;

struct Box[T] {
    v: T,
    n: int,
}

    Circle(float),
    Rect(float, float),
    Point,
}

struct Vec2 {
    x: float,
    y: float,
}

impl Vec2 {
    fn len(self) -> float {
        return std::math::sqrt(self.x * self.x + self.y * self.y);
    }
}

test "box works" {
    assert_eq(1, 1);
}

    return x;
}

fn main() -> int {
    let a: [int; 3] = [1, 2, 3];
    let p: *int = wlel_alloc(8) as *int;
    p[0] = a[0];
    a[1] += 2;
    a[2] = a[2] << 1;
    let b: Box[int] = Box { v: 1, n: 2 };
    defer std::println_str("bye");
    arena (1024) {
        let q: *void = wlel_arena();
        let r: Box[int] = Box { v: 2, n: 3 };
    }
    for i in 0..3 {
        a[i] = a[i] * 2;
    }
    for x in a {
        std::println_int(x);
    }
    while true {
        break;
    }
    if a[0] > 0 {
        return 1;
    } else if a[0] < 0 {
        return 2;
    } else {
        return 0;
    }
}
